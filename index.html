Python Flask Backend Server Code

Your Flask backend server is ready in app.py and running on http://localhost:5000/api/predict for the 6 dataset classes. 

from flask import Flask, request, jsonify
from flask_cors import CORS
from model_engine import DogSkinAIEngine

app = Flask(__name__)
CORS(app)
engine = DogSkinAIEngine()

@app.route('/api/predict', methods=['POST'])
def predict():
    file = request.files['image']
    result = engine.predict(file.read())
    return jsonify(result)

if __name__ == '__main__':
    app.run(port=5000, debug=True)
